function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); // 欄位順序: ID, Name (初始/通知名), ArriveName (抵達時填寫的名), Role, NotifyTime, ArriveTime, WaitMinutes, Status, Count sheet.appendRow([ data.id, data.name, data.arriveName || data.name, data.role, data.notifyTime, data.arriveTime, data.waitMinutes, data.status, data.count || 1 ]); return ContentService.createTextOutput(JSON.stringify({"result": "success"})) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rows = sheet.getDataRange().getValues(); var headers = rows[0]; var result = []; for (var i = 1; i < rows.length; i++) { var obj = {}; for (var j = 0; j < headers.length; j++) { obj[headers[j]] = rows[i][j]; } result.push(obj); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
